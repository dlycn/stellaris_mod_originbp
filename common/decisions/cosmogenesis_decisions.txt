# ============================================================
# 宇宙创生 5 级战争终止决议（仅对 super_player 可用）
# 一键结束所有由本帝国发起的 wg_end_threat 天灾战争（即 5 级
# 时对所有堕落帝国自动宣战的那批战争）。
# 使用与原生"撤销危机"决议相同的 end_war_effect 机制拆解战争。
# ============================================================
overpower_end_cosmo_wars = {
		potential = {
			has_country_flag = super_player
			has_ascension_perk = ap_cosmogenesis
		}
		allow = {
			has_crisis_level = crisis_cosmogenesis_level_5
			is_at_war = yes
		}
		effect = {
			every_war = {
				limit = {
					using_war_goal = {
						type = wg_end_threat
						owner = root
					}
				}
				end_war_effect = yes
			}
			custom_tooltip = overpower_end_cosmo_wars_tooltip
		}
	}
